1. **Role-Specific Relevance** (e.g., Farm Manager, Field Operator, Agronomist, Supply Chain, Finance, Compliance)

  - Include exception summary (e.g., “3 fields exceeded nitrogen threshold”)
  - Link to corrective action tracker

- **Core Fields:** Cost per ha/ac, yield projection vs. actual, soil nutrient/pH trends, compliance checkpoints, water/carbon footprint, input ROI
- **Improvements:**
  - Add predictive yield curves with confidence intervals
  - Include budget vs. actual reconciliation
  - Tag fields by certification status (Organic, GAP, Regenerative)

#### **4. Quarterly Reports** *(Strategic & ESG/Supply Chain)*
- **Core Fields:** Seasonal performance vs. historical benchmarks, resource allocation efficiency, ESG metrics (emissions, biodiversity, water stewardship), post-harvest loss, risk register updates
- **Improvements:**
  - Add scenario toggle (baseline vs. drought vs. premium market)
  - Include supply chain throughput & buyer compliance readiness
  - Export to stakeholder-ready PDF with executive summary

#### **5. Closeout Reports** *(Audit & Knowledge Capture)*
- **Core Fields:** Final yield/grade/market realization, full financial reconciliation, input utilization/waste audit, equipment lifecycle cost, lessons learned, next-cycle recommendations

function toggleFilter(cat) {

    if (activeFilters.size > 1) activeFilters.delete(cat);

class ReportsModule {
  constructor(containerId, options = {}) {
    this.container = document.getElementById(containerId);
    this.state = {
      reports: [],
      currentReport: null,
      userRole: options.userRole || 'manager',
      autosaveTimer: null,
      offlineQueue: [],
      filters: { frequency: 'all', status: 'all', query: '' }
    };
    this.init();
  }

  init() {
    this.bindEvents();
    this.loadReports();
    this.renderList();
    this.setupAutosave();
  }

  bindEvents() {
    this.container.querySelector('#btn-new-report').onclick = () => this.createNewReport();
    this.container.querySelector('#btn-export').onclick = () => this.exportReports();
    this.container.querySelector('#filter-frequency').onchange = e => { this.state.filters.frequency = e.target.value; this.renderList(); };
    this.container.querySelector('#filter-status').onchange = e => { this.state.filters.status = e.target.value; this.renderList(); };
    this.container.querySelector('#search-reports').oninput = e => { this.state.filters.query = e.target.value.toLowerCase(); this.renderList(); };
  }

  createNewReport() {
    const periodId = prompt('Enter report frequency (daily, weekly, monthly, quarterly, closeout):') || 'daily';
    const newReport = {
      id: `RPT-${Date.now()}`,
      period: periodId,
      role: this.state.userRole,
      status: 'draft',
      created: new Date().toISOString(),
      updated: new Date().toISOString(),
       {}
    };
    this.state.reports.unshift(newReport);
    this.saveState();
    this.renderList();
    this.openReport(newReport.id);
  }

  loadReports() {
    const stored = localStorage.getItem('agri_reports');
    if (stored) this.state.reports = JSON.parse(stored);
  }

  saveState() {
    localStorage.setItem('agri_reports', JSON.stringify(this.state.reports));
  }

  renderList() {
    const list = this.container.querySelector('#report-list');
    const filtered = this.state.reports.filter(r => {
      const freqMatch = this.state.filters.frequency === 'all' || r.period === this.state.filters.frequency;
      const statusMatch = this.state.filters.status === 'all' || r.status === this.state.filters.status;
      const queryMatch = !this.state.filters.query || 
        r.id.toLowerCase().includes(this.state.filters.query) || 
        r.period.toLowerCase().includes(this.state.filters.query);
      return freqMatch && statusMatch && queryMatch;
    });

    list.innerHTML = filtered.length ? filtered.map(r => `
      <li data-id="${r.id}" class="${this.state.currentReport?.id === r.id ? 'active' : ''}">
        <strong>${r.period.toUpperCase()} Report</strong>
        <span class="status status-${r.status}">${r.status}</span>
        <div class="meta">ID: ${r.id} | ${new Date(r.updated).toLocaleDateString()}</div>
      </li>
    `).join('') : '<li class="empty-state">No reports found.</li>';

    list.querySelectorAll('li').forEach(li => {
      li.onclick = () => this.openReport(li.dataset.id);
    });
  }

  openReport(id) {
    const report = this.state.reports.find(r => r.id === id);
    if (!report) return;
    this.state.currentReport = report;
    this.renderForm(report);
    this.renderList();
  }

  renderForm(report) {
    const view = this.container.querySelector('#report-view');
    const isReadonly = report.status !== 'draft';
    
    // Generate dynamic fields from framework structure (simplified for demo)
    const fields = this.generateFormFields(report.period, report.role);
    
    view.innerHTML = `
      <div class="report-form">
        <h2>${report.period.toUpperCase()} Report • ${report.role}</h2>
        <div class="form-row">
          <div class="form-group"><label>Report ID</label><input type="text" value="${report.id}" readonly></div>
          <div class="form-group"><label>Status</label><input type="text" value="${report.status.toUpperCase()}" readonly></div>
        </div>
        <div class="form-row">
          <div class="form-group"><label>Created</label><input type="text" value="${new Date(report.created).toLocaleString()}" readonly></div>
          <div class="form-group"><label>Last Updated</label><input type="text" value="${new Date(report.updated).toLocaleString()}" readonly></div>
        </div>
        ${fields.map(f => `
          <div class="form-group" style="margin:12px 0;">
            <label>${f.label}</label>
            ${f.type === 'textarea' ? `<textarea ${isReadonly ? 'readonly' : ''} data-field="${f.key}" placeholder="${f.placeholder || ''}">${report.data[f.key] || ''}</textarea>` :
              f.type === 'select' ? `<select data-field="${f.key}" ${isReadonly ? 'disabled' : ''}>${f.options.map(o => `<option value="${o}" ${report.data[f.key] === o ? 'selected' : ''}>${o}</option>`).join('')}</select>` :
              `<input type="${f.type}" data-field="${f.key}" value="${report.data[f.key] || ''}" ${isReadonly ? 'readonly' : ''} ${f.required ? 'required' : ''}>`
            }
          </div>
        `).join('')}
        <div class="form-actions">
          ${isReadonly ? 
            `<button class="btn-secondary" onclick="alert('Report locked.')">🔒 Read Only</button>` :
            `
            <button class="btn-primary" id="btn-save-draft">💾 Save Draft</button>
            <button class="btn-secondary" id="btn-submit">📤 Submit for Review</button>
            `
          }
        </div>
      </div>
    `;

    if (!isReadonly) {
      this.container.querySelector('#btn-save-draft').onclick = () => this.saveDraft();
      this.container.querySelector('#btn-submit').onclick = () => this.submitReport();
      this.container.querySelectorAll('input, textarea, select').forEach(el => {
        el.oninput = () => this.triggerAutosave();
      });
    }
  }

  generateFormFields(period, role) {
    // Maps to your PERIODS data structure. Extend as needed.
    const base = [
      { key: 'summary', label: 'Executive Summary', type: 'textarea', placeholder: 'Key highlights & deviations...' },
      { key: 'variance_notes', label: 'Variance / Root Cause', type: 'textarea', placeholder: 'Explain targets vs actuals...' },
      { key: 'actions', label: 'Next Steps / Corrective Actions', type: 'textarea', placeholder: 'Owner | Deadline | Status' },
      { key: 'compliance_check', label: 'Compliance Verified?', type: 'select', options: ['Yes', 'No', 'Partial'] }
    ];
    // Add period/role specific metrics here dynamically from your framework data
    return base;
  }

  saveDraft() {
    this.captureFormData();
    this.state.currentReport.status = 'draft';
    this.state.currentReport.updated = new Date().toISOString();
    this.saveState();
    this.renderList();
    this.toast('✅ Draft saved');
  }

  submitReport() {
    if (!confirm('Submit for review? This will lock editing.')) return;
    this.captureFormData();
    this.state.currentReport.status = 'submitted';
    this.state.currentReport.updated = new Date().toISOString();
    this.saveState();
    this.renderForm(this.state.currentReport);
    this.toast('📤 Report submitted');
    // Hook: this.api.submitReport(this.state.currentReport);
  }

  captureFormData() {
    const form = this.container.querySelector('.report-form');
    const data = {};
    form.querySelectorAll('[data-field]').forEach(el => data[el.dataset.field] = el.value);
    this.state.currentReport.data = data;
  }

  setupAutosave() {
    window.addEventListener('beforeunload', () => {
      if (this.state.currentReport?.status === 'draft') this.saveDraft();
    });
  }

  triggerAutosave() {
    clearTimeout(this.state.autosaveTimer);
    this.state.autosaveTimer = setTimeout(() => this.saveDraft(), 5000);
  }

  exportReports() {
    let csv = 'ID,Frequency,Role,Status,Created,Summary,Actions\n';
    this.state.reports.forEach(r => {
      csv += `"${r.id}","${r.period}","${r.role}","${r.status}","${new Date(r.created).toLocaleDateString()}","${(r.data.summary || '').replace(/"/g, '""')}","${(r.data.actions || '').replace(/"/g, '""')}"\n`;
    });
    const blob = new Blob([`\uFEFF${csv}`], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a'); a.href = url; a.download = `reports_export_${Date.now()}.csv`; a.click();
    URL.revokeObjectURL(url);
  }

  toast(msg) {
    const t = document.createElement('div'); t.className = 'toast'; t.textContent = msg;
    document.body.appendChild(t);
    setTimeout(() => t.remove(), 3000);
  }
}

class ReportsAPI {
  constructor(baseURL = '/api/v1') { this.baseURL = baseURL; }
  async _request(method, path, body = null) {
    const headers = { 'Content-Type': 'application/json', 'Authorization': `Bearer ${localStorage.getItem('jwt') || ''}` };
    const opts = { method, headers };
    if (body) opts.body = JSON.stringify(body);
    const res = await fetch(`${this.baseURL}${path}`, opts);
    if (!res.ok) throw new Error(`API ${res.status}: ${await res.text()}`);
    return res.json();

class SignaturePad {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d');
    this.drawing = false;
    this.lastX = 0; this.lastY = 0;
    this.setupEvents();
  }
  setupEvents() {
    const start = (e) => { e.preventDefault(); this.drawing = true; [this.lastX, this.lastY] = this.getPos(e); };
    const move = (e) => { if (!this.drawing) return; e.preventDefault(); this.drawLine(this.lastX, this.lastY, ...this.getPos(e)); };
    const end = () => { this.drawing = false; };
    this.canvas.addEventListener('pointerdown', start);
    this.canvas.addEventListener('pointermove', move);
    this.canvas.addEventListener('pointerup', end);
    this.canvas.addEventListener('pointerleave', end);
  }
  getPos(e) { const r = this.canvas.getBoundingClientRect(); return [e.clientX - r.left, e.clientY - r.top]; }
  drawLine(x1, y1, x2, y2) {
    this.ctx.beginPath(); this.ctx.moveTo(x1, y1); this.ctx.lineTo(x2, y2);
    this.ctx.strokeStyle = '#0f172a'; this.ctx.lineWidth = 2; this.ctx.lineCap = 'round'; this.ctx.stroke();
    this.lastX = x2; this.lastY = y2;
  }
  clear() { this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height); }
  getDataURL() { return this.canvas.toDataURL('image/png'); }
  isEmpty() { return this.getDataURL() === this.canvas.toDataURL('image/png', 0); }
}

### 🌐 3. Webhook Handler for WhatsApp Delivery/Read Receipts
Meta sends `POST` requests to your webhook URL for every message status change (`sent`, `delivered`, `read`, `failed`). This Express route verifies signatures, parses events, and updates your database.

#### `routes/whatsapp-webhook.js`
```javascript
const express = require('express');
const crypto = require('crypto');
const router = express.Router();

const APP_SECRET = process.env.META_APP_SECRET;
const VERIFY_TOKEN = process.env.META_VERIFY_TOKEN;
const REPORT_MODEL = require('../models/Report'); // Your DB model

// ✅ GET: Webhook verification (required by Meta)
router.get('/whatsapp', (req, res) => {
  const mode = req.query['hub.mode'];
  const token = req.query['hub.verify_token'];
  const challenge = req.query['hub.challenge'];

  if (mode === 'subscribe' && token === VERIFY_TOKEN) {
    console.log('✅ WhatsApp webhook verified');
    res.status(200).send(challenge);
  } else {
    res.sendStatus(403);
  }
});

// 🔐 Signature Verification Middleware
function verifySignature(req, res, next) {
  const signature = req.headers['x-hub-signature-256']?.split('=')[1];
  if (!signature) return res.sendStatus(401);

  const expected = crypto.createHmac('sha256', APP_SECRET)
    .update(JSON.stringify(req.body))
    .digest('hex');

  if (signature !== expected) {
    console.warn('⚠️ Invalid webhook signature');
    return res.sendStatus(403);
  }
  next();
}

// 📥 POST: Handle WhatsApp events
router.post('/whatsapp', verifySignature, async (req, res) => {
  const body = req.body;
  
  // Filter only status updates
  const statuses = body.entry?.[0]?.changes?.[0]?.value?.statuses || [];
  if (!statuses.length) return res.sendStatus(200);

  for (const status of statuses) {
    const { id: waMsgId, status: msgStatus, recipient_id } = status;
    console.log(`📡 WhatsApp Status: ${waMsgId} → ${msgStatus} | Recipient: ${recipient_id}`);

    // Map Meta status to your DB
    const dbStatus = {
      sent: 'WA_SENT',
      delivered: 'WA_DELIVERED',
      read: 'WA_READ',
      failed: 'WA_FAILED'
    }[msgStatus] || 'UNKNOWN';

    try {
      // Update report/notification log (adjust to your schema)
      await REPORT_MODEL.updateOne(
        { whatsapp_message_id: waMsgId },
        { $set: { whatsapp_status: dbStatus, last_status_update: new Date() } }
      );
    } catch (err) {
      console.error('❌ DB update failed for WA receipt:', err.message);
    }
  }

  res.sendStatus(200);
});

module.exports = router;
```

### 🔄 2. Retry Queue with Exponential Backoff (BullMQ + Redis)

#### A. Install Dependencies

#### B. Queue Definition (`queues/whatsappRetry.js`)

#### C. Worker with Backoff (`workers/retryWorker.js`)
```javascript
const { Worker } = require('bullmq');
const IORedis = require('ioredis');
const { notifyOnSubmit, notifyOnReview } = require('../notificationServiceMeta');

const connection = new IORedis(process.env.REDIS_URL || 'redis://localhost:6379', { maxRetriesPerRequest: null });

const worker = new Worker('whatsapp-retry', async job => {
  const { type, recipient, data, attempt } = job.data;
  console.log(`🔄 Retry attempt ${attempt} for ${recipient} (${type})`);

  try {
    if (type === 'submit') await notifyOnSubmit(data.report, data.reviewers);
    else if (type === 'review') await notifyOnReview(data.report);
    console.log(`✅ Retry successful for ${recipient}`);
  } catch (err) {
    console.error(`❌ Retry failed for ${recipient}:`, err.message);
    throw err; // BullMQ handles backoff
  }
}, {
  connection,
  limiter: { max: 10, duration: 60000 } // Rate limit to 10/min per queue
});

worker.on('failed', (job, err) => {
  if (job.attemptsMade >= 5) {
    console.error(`📦 Dead-letter: ${job.data.recipient} after ${job.attemptsMade} attempts`);
    // Push to DB dead-letter table for manual review
  }
});

module.exports = worker;
```

#### D. Integrate into Notification Flow
Replace direct `await sendWhatsApp()` with queue fallback:
```javascript
async function sendWithRetry(type, recipient, data) {
  try {
    if (type === 'submit') await notifyOnSubmit(data.report, data.reviewers);
    else await notifyOnReview(data.report);
  } catch (err) {
    await retryQueue.add('retry', { type, recipient, data }, {
      attempts: 5,
      backoff: { type: 'exponential', delay: 2000 } // 2s, 4s, 8s, 16s, 32s
    });
  }
}
```
